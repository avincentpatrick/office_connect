# --- Attachments service (Increment 4) ---
# Pillow re-encodes uploaded images and strips EXIF/XMP; pillow-heif decodes
# HEIC/HEIF (iPhone photos) so they can be normalized to JPEG. Both ship
# self-contained manylinux wheels (libjpeg/zlib/libheif bundled) — no apt libs.
# clamd is a pure-Python TCP client for a ClamAV daemon (lazy-imported; the
# scanner is injectable and fail-closed, so ClamAV is opt-in via a compose profile).
Pillow==11.1.0
pillow-heif==0.21.0
clamd==1.0.2
# multipart/form-data parsing for the Stage-B attachments upload + directory CSV
# import routers (FastAPI requires it whenever a route declares File()/Form()).

# --- Observability (Increment 4) ---
# Self-hosted error tracking (GlitchTip is Sentry-SDK compatible). Lazy-imported
# and only active when SENTRY_DSN is set — no DSN = fully no-op. Structured JSON
# logging uses the stdlib only (core/logging.py), no extra dependency.
sentry-sdk==2.20.0

# --- Auth / identity (Stage B / Phase 2) ---
# Argon2id password hashing — OWASP first-choice (memory-hard, GPU-resistant).
# argon2-cffi ships self-contained manylinux wheels (no apt libs). Parameters are
# recorded in docs/standards/tech-stack.md (standing rule 9); the stored PHC hash
# self-describes its cost so raising it later re-hashes on next login.
argon2-cffi==23.1.0
# TOTP MFA (RFC 6238) for approver/admin roles (Increment 2; NPC Circular 2023-06).
# Pure-Python (no OS libs); returns the otpauth:// provisioning URI so the client
# renders the QR — no QR-image dependency needed server-side.
pyotp==2.9.0

# --- Document generation (Stage C R-5 / core-service #8) ---
# Jinja2 renders the government forms to HTML; WeasyPrint turns that HTML+CSS
# into print-faithful PDFs (A4, @page running footers). Chosen over wkhtmltopdf
# (archived Jan 2023, ancient WebKit) and over a headless browser: WeasyPrint is
# pure Python and needs no Chromium on an offline on-prem server.
# WeasyPrint needs Pango/HarfBuzz/libffi at RUNTIME — see the Dockerfile apt
# layer and docs/standards/tech-stack.md §3. It is imported LAZILY inside
# core/documents/render.py so importing core never requires the native stack
# (that is what keeps the test suite runnable on a Windows host).
# RULE: rendering happens in the Celery worker, never in the request path.
Jinja2==3.1.5
weasyprint==63.1
